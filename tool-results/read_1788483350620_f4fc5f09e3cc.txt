   314→# === Fonction Prophet ===
   315→def fit_prophet_fast(history_df, periods=12, freq='MS'):
   316→    """Fit Prophet and forecast 12 months starting from Jan 2027."""
   317→    future_dates = pd.date_range(start='2027-01-01', periods=periods, freq=freq)
   318→    if len(history_df) < 4:
   319→        avg = history_df['y'].mean() if len(history_df) > 0 else 0
   320→        return pd.DataFrame({'ds': future_dates, 'yhat': [avg] * periods,
   321→                             'yhat_lower': [avg * 0.8] * periods, 'yhat_upper': [avg * 1.2] * periods})
   322→    try:
   323→        m = Prophet(yearly_seasonality=True, weekly_seasonality=False, daily_seasonality=False,
   324→                    seasonality_mode='multiplicative', changepoint_prior_scale=0.05,
   325→                    interval_width=0.8, mcmc_samples=0)
   326→        m.fit(history_df[['ds', 'y']])
   327→        future = pd.DataFrame({'ds': future_dates})
   328→        forecast = m.predict(future)
   329→        return forecast[['ds', 'yhat', 'yhat_lower', 'yhat_upper']]
   330→    except Exception as e:
   331→        print(f"    Prophet error: {e}")
   332→        avg = history_df['y'].mean()
   333→        return pd.DataFrame({'ds': future_dates, 'yhat': [avg] * periods,
   334→                             'yhat_lower': [avg * 0.8] * periods, 'yhat_upper': [avg * 1.2] * periods})

   336→def extrapolate_mean(history_df, periods=12, freq='MS', value_col='y', family=None):
   337→    """Extrapolate using historical monthly average for 2027.
   338→    For ALVEOLES: use 2026 monthly average (very low) instead of 2025 spike."""
   339→    future_dates = pd.date_range(start='2027-01-01', periods=periods, freq=freq)
   340→    
   341→    # SPECIAL CASE: ALVEOLES — use 2026 monthly average (very low) instead of 2025 spike
   342→    if family == 'ALVEOLES':
   343→        # 2026 ALVEOLES: 11 records, 13.9 M HT over 8 months → ~1.7 M/month
   344→        avg_2026 = history_df[history_df['ds'].dt.year == 2026][value_col].mean()
   345→        if pd.isna(avg_2026) or avg_2026 == 0:
   346→            avg_2026 = 1.5  # Fallback: ~1.5 M/month
   347→        # Apply seasonal weights (Q4 slight uptick for year-end)
   348→        weights = [0.8, 0.9, 1.0, 1.0, 1.1, 1.0, 0.9, 0.9, 0.9, 1.2, 1.1, 1.0]
   349→        return pd.DataFrame({
   350→            'ds': future_dates,
   351→            'yhat': [avg_2026 * w for w in weights],
   352→            'yhat_lower': [avg_2026 * w * 0.7 for w in weights],
   353→            'yhat_upper': [avg_2026 * w * 1.3 for w in weights],
   354→        })
   355→    
   356→    avg = history_df[value_col].mean() if len(history_df) > 0 else 0
   357→    return pd.DataFrame({'ds': future_dates, 'yhat': [avg] * periods,
