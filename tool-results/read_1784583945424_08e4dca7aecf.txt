   237→# ===== HELPER FUNCTIONS (identical to build_pdf_report_v3.py) =====
   238→def make_table(data, col_widths=None, header_row=True, font_size=9, align='LEFT'):
   239→    """Create a styled table with BELGOCAM colors. Auto-scales to full content width."""
   240→    if col_widths is None:
   241→        n_cols = len(data[0])
   242→        col_widths = [CONTENT_W / n_cols] * n_cols
   243→    else:
   244→        # Auto-scale: if sum of col_widths < CONTENT_W, scale up proportionally
   245→        total = sum(col_widths)
   246→        if total < CONTENT_W * 0.95:
   247→            scale = CONTENT_W / total
   248→            col_widths = [w * scale for w in col_widths]
   249→
   250→    # Wrap text cells in Paragraphs for proper wrapping
   251→    cell_style = ParagraphStyle('cell_fs', parent=CELL, fontSize=font_size, leading=font_size + 3)
   252→    cell_style_r = ParagraphStyle('cell_r_fs', parent=CELL_RIGHT, fontSize=font_size, leading=font_size + 3)
   253→    cell_style_c = ParagraphStyle('cell_c_fs', parent=CELL_CENTER, fontSize=font_size, leading=font_size + 3)
   254→    cell_style_w = ParagraphStyle('cell_w_fs', parent=CELL_WHITE, fontSize=font_size, leading=font_size + 3)
   255→
   256→    wrapped = []
   257→    for i, row in enumerate(data):
   258→        wrapped_row = []
   259→        for j, cell in enumerate(row):
   260→            if isinstance(cell, str):
   261→                if i == 0 and header_row:
   262→                    style = cell_style_w
   263→                else:
   264→                    style = cell_style_r if align == 'RIGHT' else cell_style
   265→                wrapped_row.append(Paragraph(cell, style))
   266→            else:
   267→                wrapped_row.append(cell)
   268→        wrapped.append(wrapped_row)
   269→
   270→    t = Table(wrapped, colWidths=col_widths, repeatRows=1 if header_row else 0)
   271→    ts = TableStyle([
   272→        ('VALIGN', (0, 0), (-1, -1), 'MIDDLE'),
   273→        ('LEFTPADDING', (0, 0), (-1, -1), 5),
   274→        ('RIGHTPADDING', (0, 0), (-1, -1), 5),
   275→        ('TOPPADDING', (0, 0), (-1, -1), 3),
   276→        ('BOTTOMPADDING', (0, 0), (-1, -1), 3),
   277→        ('GRID', (0, 0), (-1, -1), 0.4, GRAY_LIGHT),
   278→    ])
   279→    if header_row:
   280→        ts.add('BACKGROUND', (0, 0), (-1, 0), NAVY)
   281→        ts.add('TEXTCOLOR', (0, 0), (-1, 0), WHITE)
   282→        ts.add('FONTNAME', (0, 0), (-1, 0), 'NotoSerifSC-Bold')
   283→    # Banding
   284→    for i in range(1, len(data)):
   285→        if i % 2 == 0:
   286→            ts.add('BACKGROUND', (0, i), (-1, i), GRAY_VLIGHT)
   287→    t.setStyle(ts)
   288→    return t

   291→def kpi_card(label, value, sublabel=""):
   292→    """Create a small KPI card as a Table."""
   293→    data = [
   294→        [Paragraph(f'<b>{value}</b>', ParagraphStyle('kpi_v', fontName='NotoSerifSC-Bold', fontSize=13, textColor=NAVY, alignment=TA_CENTER, leading=16))],
   295→        [Paragraph(label, ParagraphStyle('kpi_l', fontName='NotoSerifSC', fontSize=8, textColor=GRAY, alignment=TA_CENTER, leading=10))],
   296→    ]
   297→    if sublabel:
   298→        data.append([Paragraph(sublabel, ParagraphStyle('kpi_s', fontName='NotoSerifSC-Light', fontSize=7, textColor=GRAY, alignment=TA_CENTER, leading=9))])
   299→    t = Table(data, colWidths=[3.2*cm])
   300→    t.setStyle(TableStyle([
   301→        ('BACKGROUND', (0, 0), (-1, -1), GRAY_VLIGHT),
   302→        ('LINEBELOW', (0, 0), (-1, 0), 0.5, GOLD),
   303→        ('LEFTPADDING', (0, 0), (-1, -1), 4),
   304→        ('RIGHTPADDING', (0, 0), (-1, -1), 4),
   305→        ('TOPPADDING', (0, 0), (-1, -1), 4),
   306→        ('BOTTOMPADDING', (0, 0), (-1, -1), 4),
   307→        ('BOX', (0, 0), (-1, -1), 0.3, GRAY_LIGHT),
   308→    ]))
   309→    return t

   318→        ('VALIGN', (0, 0), (-1, -1), 'TOP'),
   319→        ('LEFTPADDING', (0, 0), (-1, -1), 4),
   320→        ('RIGHTPADDING', (0, 0), (-1, -1), 4),
   321→    ]))
   322→    return t
