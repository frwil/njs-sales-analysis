   271→def make_table(data, col_widths=None, header_row=True, font_size=9, align='LEFT'):
   272→    """Create a styled table with BELGOCAM colors. Auto-scales to full content width."""
   273→    if col_widths is None:
   274→        n_cols = len(data[0])
   275→        col_widths = [CONTENT_W / n_cols] * n_cols
   276→    else:
   277→        # Auto-scale: if sum of col_widths < CONTENT_W, scale up proportionally
   278→        total = sum(col_widths)
   279→        if total < CONTENT_W * 0.95:
   280→            scale = CONTENT_W / total
   281→            col_widths = [w * scale for w in col_widths]
   282→
   283→    # Wrap text cells in Paragraphs for proper wrapping
   284→    # Use a font-size-aware cell style
   285→    cell_style = ParagraphStyle('cell_fs', parent=CELL, fontSize=font_size, leading=font_size + 3)
   286→    cell_style_r = ParagraphStyle('cell_r_fs', parent=CELL_RIGHT, fontSize=font_size, leading=font_size + 3)
   287→    cell_style_c = ParagraphStyle('cell_c_fs', parent=CELL_CENTER, fontSize=font_size, leading=font_size + 3)
   288→    cell_style_w = ParagraphStyle('cell_w_fs', parent=CELL_WHITE, fontSize=font_size, leading=font_size + 3)
   289→
   290→    wrapped = []
   291→    for i, row in enumerate(data):
   292→        wrapped_row = []
   293→        for j, cell in enumerate(row):
   294→            if isinstance(cell, str):
   295→                if i == 0 and header_row:
   296→                    style = cell_style_w
   297→                else:
   298→                    style = cell_style_r if align == 'RIGHT' else cell_style
   299→                wrapped_row.append(Paragraph(cell, style))
   300→            else:
   301→                wrapped_row.append(cell)
   302→        wrapped.append(wrapped_row)
   303→
   304→    t = Table(wrapped, colWidths=col_widths, repeatRows=1 if header_row else 0)
   305→    ts = TableStyle([
   306→        ('VALIGN', (0, 0), (-1, -1), 'MIDDLE'),
   307→        ('LEFTPADDING', (0, 0), (-1, -1), 5),
   308→        ('RIGHTPADDING', (0, 0), (-1, -1), 5),
   309→        ('TOPPADDING', (0, 0), (-1, -1), 3),
   310→        ('BOTTOMPADDING', (0, 0), (-1, -1), 3),
   311→        ('GRID', (0, 0), (-1, -1), 0.4, GRAY_LIGHT),
   312→    ])
   313→    if header_row:
   314→        ts.add('BACKGROUND', (0, 0), (-1, 0), NAVY)
   315→        ts.add('TEXTCOLOR', (0, 0), (-1, 0), WHITE)
   316→        ts.add('FONTNAME', (0, 0), (-1, 0), 'NotoSerifSC-Bold')
   317→    # Banding
   318→    for i in range(1, len(data)):
   319→        if i % 2 == 0:
   320→            ts.add('BACKGROUND', (0, i), (-1, i), GRAY_VLIGHT)
   321→    t.setStyle(ts)
   322→    return t

   363→def heading_with_content(heading_text, content_flowables, level=2):
   364→    """Create a heading that stays together with its first content element.
   365→    Prevents orphan headings at the bottom of pages.
   366→    level=2 for H2, level=3 for H3.
   367→    """
   368→    style = H2 if level == 2 else H3
   369→    heading = Paragraph(heading_text, style)
   370→    # Keep heading + first content together
   371→    if isinstance(content_flowables, list):
   372→        return KeepTogether([heading] + content_flowables[:2])
   373→    else:
   374→        return KeepTogether([heading, content_flowables])
